# 添加数字签名

## 概述

+ 为了使用API更加安全，有时候需要后端进行数字签名，即： key + sig 方式

  ```javascript
  // 前端
  const location = '116.310003,39.991957'
  const url = `/users?location=${location}`

  fetch(url)
    .then(response => response.json())
    .then(data => {
      console.log('Geocode data:', data)
    })
    .catch(error => {
      console.error('Fetch error:', error)
    })
  ```

  ```javascript
  // 后端

  var express = require('express');
  var router = express.Router();
  const axios = require('axios');
  const crypto = require('crypto');


  const PRIVATE_KEY = '27609d83402fae5b4dea68c1a5fd091d'
  const GD_TOKEN = 'b0a6dd06cdc2be8bfee5a5ec37b0a131'
  const url = 'https://restapi.amap.com/v3/geocode/regeo'

  router.get('/', async function (req, res, next) {
    try {
      // 前端传 location，例如 ?location=116.310003,39.991957
      const { location } = req.query;

      if (!location) {
        return res.status(400).json({ error: '缺少 location 参数' });
      }

      // 1. 准备请求参数
      const params = {
        key: GD_TOKEN,
        location,
        output: 'json'
      };

      // 2. 拼接参数（升序）
      const sortedParams = Object.keys(params)
        .sort()
        .map(k => `${k}=${params[k]}`)
        .join('&');

      // 3. 拼接私钥
      const strToSign = sortedParams + PRIVATE_KEY;

      // 4. 生成 sig
      const sig = crypto.createHash('md5').update(strToSign).digest('hex');

      // 5. 发起请求
      const response = await axios.get(url, {
        params: {
          ...params,
          sig
        }
      });

      res.json(response.data);

    } catch (error) {
      res.status(500).json({ error: error.message });
    }
  });

  module.exports = router;

  ```
