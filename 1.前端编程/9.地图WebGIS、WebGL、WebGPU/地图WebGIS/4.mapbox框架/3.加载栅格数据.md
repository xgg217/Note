# 加载栅格数据

## 概述

+ 支持静态图片添加到地图

  ```javascript
  import xc from './assets/xc.png'

  const extent = [
    [101.73497559, 27.50030558],
    [102.43557243, 28.18097996],
  ]

  map.on('load', () => {
    map.addSource('img_xc', {
      type: 'image',
      url: xc,
      coordinates: [
        [extent[0][0], extent[0][1]], // 左下角
        [extent[1][0], extent[0][1]], // 右下角
        [extent[1][0], extent[1][1]], // 右上角
        [extent[0][0], extent[1][1]], // 左上角
      ],
    })
    map.addLayer({
      id: 'layer_xc',
      type: 'raster',
      source: 'img_xc',
    })
    map.fitBounds(extent, {
      padding: { top: 20, bottom: 20, left: 20, right: 20 },
    })
  })
  ```
