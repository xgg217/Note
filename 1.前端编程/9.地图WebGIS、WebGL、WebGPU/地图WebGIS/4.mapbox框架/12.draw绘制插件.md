# draw绘制插件

### 概述

+ 添加了对 Mapbox GL JS 地图上的绘图和编辑功能的支持

  ```shell
  npm install @mapbox/mapbox-gl-draw
  ```

  ```javascript
  import MapboxDraw from '@mapbox/mapbox-gl-draw'
  import '@mapbox/mapbox-gl-draw/dist/mapbox-gl-draw.css'
  import { area, convertArea, length } from '@turf/turf'

  const draw = new MapboxDraw()
  map.addControl(draw, 'top-left')

  map.on('draw.create', (e) => {

    const data = draw.getAll()
    if (data.features[0].geometry.type === 'Polygon') {
      const areaData = convertArea(area(data), 'meters', 'kilometers')  // 默认平方米, 转换成平方公里
      console.log('draw.create', areaData)
    } else if (data.features[0].geometry.type === 'LineString') {
      const lineData = length(data)  // 默认公里
      console.log('draw.create', lineData)
    }

  })
  ```
