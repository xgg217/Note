# 与地图框架结合2

## 概述

+ 获取到 Web服务 API 后，可以与任意地图框架结合，例如：OpenLayers、Leaflet、Mapbox等

  ```vue
  <script setup>
    import { onMounted, ref, computed } from 'vue'
    import { Map, View } from 'ol'
    import { Tile as TileLayer, Vector as VectorLayer } from 'ol/layer'
    import { Vector } from 'ol/source'
    import { XYZ } from 'ol/source'
    import { fromLonLat } from 'ol/proj'
    import { Style, Icon } from 'ol/style'
    import { GeoJSON } from 'ol/format'
    import center from './assets/center.svg'

    let map
    const TDT_TOKEN = '55c0066c8dd3630394c756dfcd97add5'
    const GD_TOKEN = 'fd7a0e4975078b5d697d478286ce46a8'

    const citys = [
      { name: '北京', coord: [116.405285, 39.904989] },
      { name: '上海', coord: [121.472644, 31.231706] },
      { name: '广州', coord: [113.280637, 23.125178] },
      { name: '深圳', coord: [114.085947, 22.547] },
      { name: '杭州', coord: [120.153576, 30.287459] },
    ]
    const searchWord = ref('KTV')
    const selectCity = ref(citys[0].name)
    const url = computed(() => {
      return `https://restapi.amap.com/v5/place/text?key=${GD_TOKEN}&types=${searchWord.value}&region=${selectCity.value}`
    })

    const searchLayer = new VectorLayer({
      source: new Vector({
        features: [],
      }),
      style: new Style({
        image: new Icon({
          src: center,
          scale: 0.2,
        }),
      }),
    })

    const json2Geojson = (json) => {
      const result = { type: 'FeatureCollection', features: [] }
      json.forEach((j) => {
        const lnglat = j.location.split(',')
        const feature = {
          type: 'Feature',
          properties: {},
          geometry: { type: 'Point', coordinates: [] },
        }
        feature.properties = j
        feature.geometry.coordinates = lnglat
        result.features.push(feature)
      })
      return result
    }

    const handleToSearch = () => {
      fetch(url.value)
        .then(response => response.json())
        .then(data => {
          // console.log('Geocode data:', data)

          const features = new GeoJSON().readFeatures(json2Geojson(data.pois), {
            featureProjection: 'EPSG:3857',
            dataProjection: 'EPSG:4326',
          })
          searchLayer.getSource().clear()
          searchLayer.getSource().addFeatures(features)

          const extent = searchLayer.getSource().getExtent()
          if (extent[0] !== Infinity) {
            map.getView().fit(extent, { size: map.getSize(), padding: [50, 50, 50, 50] })
          }

        })
        .catch(error => {
          console.error('Fetch error:', error)
        })
    }

    onMounted(() => {
      map = new Map({
        target: 'map',
        layers: [
          new TileLayer({
            source: new XYZ({
              url: 'http://t0.tianditu.gov.cn/vec_w/wmts?SERVICE=WMTS&REQUEST=GetTile&VERSION=1.0.0&LAYER=vec&STYLE=default&TILEMATRIXSET=w&FORMAT=tiles&TILEMATRIX={z}&TILEROW={y}&TILECOL={x}&tk=' + TDT_TOKEN
            }),
          }),
          new TileLayer({
            source: new XYZ({
              url: 'http://t0.tianditu.gov.cn/cva_w/wmts?SERVICE=WMTS&REQUEST=GetTile&VERSION=1.0.0&LAYER=cva&STYLE=default&TILEMATRIXSET=w&FORMAT=tiles&TILEMATRIX={z}&TILEROW={y}&TILECOL={x}&tk=' + TDT_TOKEN
            }),
          }),
          searchLayer,
        ],
        view: new View({
          center: fromLonLat(citys[0].coord),
          zoom: 10,
        })
      })
    })
  </script>

  <template>
    <div id="map"></div>
    <el-input class="search" v-model="searchWord" style="width: 400px" placeholder="搜索关键词">
      <template #prepend>
        <el-select v-model="selectCity" style="width: 100px">
          <el-option v-for="item in citys" :key="item.name" :label="item.name" :value="item.name" />
        </el-select>
      </template>
      <template #append>
        <el-button icon="Search" @click="handleToSearch" />
      </template>
    </el-input>
  </template>

  <style scoped>
  #map {
    position: absolute;
    inset: 0;
    overflow: hidden;
  }

  .search {
    position: absolute;
    top: 10px;
    left: 50px;
    z-index: 1;
  }
  </style>

  ```
