# 前端加载WFS标准

## 概述

+ GeoServer发布的WFS标准协议接口，可以通过前端地图框架（OpenLayers、Mapbox、Leaflet等）进行加载

  ```javascript
  import { Map, View } from 'ol'
  import { Tile as TileLayer, Vector as VectorLayer } from 'ol/layer'
  import { XYZ, Vector } from 'ol/source'
  import { fromLonLat } from 'ol/proj'
  import { GeoJSON } from 'ol/format'

  const TDT_TOKEN = '55c0066c8dd3630394c756dfcd97add5'
  const coodPoint = fromLonLat([120.22, 30.26])

  const vectorSource = new Vector({
    url: '/geoserver/ne/ows?service=WFS&version=1.0.0&request=GetFeature&typeName=ne%3Apoints&outputFormat=application%2Fjson&maxFeatures=50',
    format: new GeoJSON({
      dataProjection: 'EPSG:4326',
      featureProjection: 'EPSG:3857',
    }),
  })

  const vectorLayer = new VectorLayer({
    source: vectorSource,
    style: {
      'circle-radius': 10,
      'circle-fill-color': '#FF0000',
    },
  })

  const map = new Map({
    target: 'map',
    layers: [
      new TileLayer({
        source: new XYZ({
          url: 'http://t0.tianditu.gov.cn/vec_w/wmts?SERVICE=WMTS&REQUEST=GetTile&VERSION=1.0.0&LAYER=vec&STYLE=default&TILEMATRIXSET=w&FORMAT=tiles&TILEMATRIX={z}&TILEROW={y}&TILECOL={x}&tk=' + TDT_TOKEN
        }),
      }),
      new TileLayer({
        source: new XYZ({
          url: 'http://t0.tianditu.gov.cn/cva_w/wmts?SERVICE=WMTS&REQUEST=GetTile&VERSION=1.0.0&LAYER=cva&STYLE=default&TILEMATRIXSET=w&FORMAT=tiles&TILEMATRIX={z}&TILEROW={y}&TILECOL={x}&tk=' + TDT_TOKEN
        }),
      }),
      vectorLayer,
    ],
    view: new View({
      center: coodPoint,
      zoom: 5,
    })
  })

  map.on('singleclick', function (e) {
    map.forEachFeatureAtPixel(e.pixel, function (feature, layer) {
      if (layer === vectorLayer) {
        alert('点击了点要素: ' + feature.get('name'))
      }
    })
  })
  ```
