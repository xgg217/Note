# 调用服务插件

## 概述

+ 在 高德 JS API 2.0 中，服务插件就是各种功能服务类，例如：搜索、规划等

  ```vue
  <script setup>
    AMap.plugin(["AMap.PlaceSearch"], function () {
      const placeSearch = new AMap.PlaceSearch({
        pageSize: 5,
        pageIndex: 1,
        city: "010",
        citylimit: true,
        map: map,
        panel: "my-panel",
        autoFitView: true,
      })
      placeSearch.search("北京大学")   // 使用插件搜索关键字并查看结果
    })
  </script>
  <template>
    <div id="map"></div>
    <div id="my-panel"></div>
  </template>
  <style scoped>
  #map {
    position: absolute;
    inset: 0;
  }

  #my-panel {
    position: absolute;
    top: 10px;
    right: 10px;
    width: max-content;
    max-height: 500px;
    overflow: auto;
    background: white;
    z-index: 100;
    padding: 5px;
    box-shadow: 0 0 5px rgba(0, 0, 0, 0.3);
  }
  </style>
  ```

  ```vue
  <script setup>
  const points = [
    { keyword: '北京市地震局（公交站）', city: '北京' },
    { keyword: '亦庄文化园（地铁站）', city: '北京' },
  ]

  AMap.plugin(["AMap.Driving"], function () {
    const driving = new AMap.Driving({
      map: map,
      panel: "my-panel",
    })

    driving.search(points, function (status, result) {
      if (status === "complete") {
        console.log("驾车导航信息：" + result)
      } else {
        console.log("获取驾车数据失败：" + result)
      }
    })
  })
  </script>
  ...
  ```
