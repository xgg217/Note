# L7加载GeoJSON

## 概述

+ L7支持的数据格式有GeoJSON，CSV，JSON，Image

  ```javascript
  import hangzhou from './assets/hangzhou.json'

  const polyLayer = new PolygonLayer()
    .source(hangzhou)
    .color('#FFFF00')
    .style({
      opacity: 0.8,
    })
  const lineLayer = new LineLayer()
    .source(hangzhou)
    .color('#FF0000')
    .size(1)
  const pointLayer = new PointLayer()
    .source(hangzhou)
    .shape('name')
    .color('#0000FF')
    .size(20)
    .style({
      opacity: 1.0,
      stroke: '#FFFFFF',
      strokeWidth: 1,
    })

  scene.addLayer(polyLayer)
  scene.addLayer(lineLayer)
  scene.addLayer(pointLayer)

  polyLayer.fitBounds()
  ```

  ```javascript
  const hangzhou_url = 'https://geo.datav.aliyun.com/areas_v3/bound/geojson?code=330100_full'

  fetch(hangzhou_url)
    .then((res) => res.json())
    .then((data) => {
      const polyLayer = new PolygonLayer()
        .source(data)
        .color('#FFFF00')
        .style({
          opacity: 0.8,
        })
      const lineLayer = new LineLayer()
        .source(data)
        .color('#FF0000')
        .size(1)
      const pointLayer = new PointLayer()
        .source(data)
        .shape('name')
        .color('#0000FF')
        .size(20)
        .style({
          opacity: 1.0,
          stroke: '#FFFFFF',
          strokeWidth: 1,
        })

      scene.addLayer(polyLayer)
      scene.addLayer(lineLayer)
      scene.addLayer(pointLayer)

      polyLayer.fitBounds()
    })
  ```
