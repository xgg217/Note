# 实现热力图

## 概述

+ 热力图（Heatmap） 是一种用颜色的渐变和强度来表示点数据的密度或权重分布的可视化方式

  ```javascript
  import tentea from './assets/tentea.json'

  map.addLayer({
    id: 'layer_tentea',
    type: 'heatmap',
    source: 'geojson_tentea',
    layout: {},
    paint: {
      // 点的权重
      "heatmap-weight": [
        "interpolate",
        ["linear"],
        ["get", "temp"],
        15.38,
        0,
        20.45,
        1,
      ],
      // 热力强度
      "heatmap-intensity": [
        "interpolate",
        ["linear"],
        ["zoom"],
        0,
        1,
        19,
        13,
      ],
      // 热力颜色梯度
      "heatmap-color": [
        "interpolate",
        ["linear"],
        ["heatmap-density"],
        0,
        "rgba(255, 254, 191,0)",
        0.2,
        "rgb(91, 182, 255)",
        0.4,
        "rgb(0, 48, 255)",
        0.6,
        "rgb(255, 202, 114)",
        0.8,
        "rgb(255, 121, 60)",
        1,
        "rgb(255, 18, 5)",
      ],
      // 热力图的半径
      "heatmap-radius": [
        "interpolate",
        ["linear"],
        ["zoom"],
        0,
        15, 19, 110
      ],
      // 热力透明度
      "heatmap-opacity": [
        "interpolate",
        ["linear"],
        ["zoom"],
        7,
        1, 19, 0
      ],
    },
  })
  ```
