# 添加GeoJSON数据

## 概述

+ 支持GeoJSON数据添加到地图

  ```javascript
  import { bbox } from '@turf/turf'
  import hangzhou from './assets/hangzhou.json'

  map.addSource('geojson_hz', {
    type: 'geojson',
    data: hangzhou,
  })

  map.addLayer({
    id: 'layer_hz',
    type: 'fill',
    source: 'geojson_hz',
    layout: {},
    paint: {
      'fill-color': '#FFFF00',
      'fill-opacity': 0.5,
    },
  })

  map.addLayer({
    id: 'layer_hz2',
    type: 'line',
    source: 'geojson_hz',
    layout: {},
    paint: {
      'line-color': '#FF0000',
      'line-width': 1,
    },
  })

  map.fitBounds(bbox(hangzhou), {
    padding: { top: 20, bottom: 20, left: 20, right: 20 },
  })
  ```
